cmake_minimum_required(VERSION 3.24)
project(HS2ManagerFbxBridge LANGUAGES CXX)
set(FBX_SDK_ROOT "" CACHE PATH "Unmodified official Autodesk FBX SDK directory")
if(NOT EXISTS "${FBX_SDK_ROOT}/include/fbxsdk.h")
  message(FATAL_ERROR "FBX_SDK_ROOT must contain the official SDK headers and libraries")
endif()
add_library(HS2Manager.FbxBridge SHARED FbxBridge.cpp)
target_compile_features(HS2Manager.FbxBridge PRIVATE cxx_std_17)
target_compile_definitions(HS2Manager.FbxBridge PRIVATE FBXSDK_SHARED)
target_include_directories(HS2Manager.FbxBridge PRIVATE "${FBX_SDK_ROOT}/include")
target_link_libraries(HS2Manager.FbxBridge PRIVATE "${FBX_SDK_ROOT}/lib/x64/release/libfbxsdk.lib")
if(MSVC)
  target_compile_options(HS2Manager.FbxBridge PRIVATE /W4 /EHsc)
endif()
add_custom_command(TARGET HS2Manager.FbxBridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${FBX_SDK_ROOT}/lib/x64/release/libfbxsdk.dll" "$<TARGET_FILE_DIR:HS2Manager.FbxBridge>")
